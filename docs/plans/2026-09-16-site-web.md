> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Construire le site statique d'un ostéopathe (accueil, page Vendenheim, page Schiltigheim, mentions légales) qui convertit les visiteurs en rendez-vous Doctolib et se positionne sur les recherches locales.

**Architecture:** Site Astro 100 % statique dans `site/`. **Tech Stack:** Node 24, Astro (dernière version), @astrojs/sitemap, @fontsource-variable (Archivo, Source Sans 3), sharp (génération d'images), html-validate, `node --test` (TypeScript exécuté nativement par Node 24).

**Spec:** [docs/specs/2026-09-16-site-web-design.md](../specs/2026-09-16-site-web-design.md)

## Global Constraints

- Tout le code du site vit dans `site/`. Toutes les commandes `npm` se lancent depuis `site/` ; toutes les commandes `git` depuis la racine du dépôt (`siteweb/`).
- Astro échappe les attributs (`&` peut devenir `&#38;` ou `&amp;`) : les tests comparent le HTML après `decoder()`.
- Langue : français, **vouvoiement** des visiteurs, textes du praticien à la première personne.
- Accords neutres dans les textes du praticien : pas d'adjectif ou participe genré se rapportant à lui (« diplômé », « installé », « passionné » interdits ; écrire « formation à », « depuis », « ma passion »).
- Aucune promesse de guérison : les mots « guérir », « guérison », « miracle » sont interdits dans les contenus.
- Aucun tiret cadratin (—) dans les textes affichés.
- Couleurs uniquement via les jetons : nuit `#14284B`, acier `#2F6FDB`, acier foncé `#1F57B8`, fond `#F7F8FA`, surface `#FFFFFF`, texte `#5B6475`, ligne `#DCE1EA`. Aucune valeur de couleur écrite en dur dans les composants.
- Contraste WCAG AA (≥ 4,5) pour tout texte ; le bleu acier n'est utilisé que comme fond de bouton avec texte blanc ; les liens texte utilisent l'acier foncé.
- Aucun cookie, aucun script tiers, aucune ressource chargée depuis un autre domaine. Seuls des **liens** sortants sont autorisés (Doctolib, Google Maps, fiches Google).
- Polices hébergées localement via @fontsource-variable.
- Élément manquant = chaîne exacte `[À FOURNIR]` (constante `A_FOURNIR`). Le script `verifier:lancement` bloque la mise en ligne tant qu'il en reste.
- Domaine : `https://lejeune-osteopathe.fr`, URL avec barre finale (`/osteopathe-vendenheim/`).
- Coordonnées : téléphone `07 61 77 40 82` (`+33761774082`), e-mail `lejeuneguillaume5@gmail.com`, Doctolib `https://www.doctolib.fr/osteopathe/strasbourg/guillaume-lejeune`.

```ts
export function lirePage(chemin: string): string {

export function extraire(html: string, regex: RegExp): string[] {
  return [...html.matchAll(regex)].map((m) => m[1]);

export function decoder(texte: string): string {
  return texte
    .replaceAll('&#39;', "'")
    .replaceAll('&#34;', '"')
    .replaceAll('&quot;', '"')
    .replaceAll('&#38;', '&')
    .replaceAll('&lt;', '<')
    .replaceAll('&gt;', '>')
    .replaceAll('&amp;', '&');
}
```

- [ ] **Step 5: Créer `site/src/lib/format.ts`**

```ts
import type { Jour, Lieu } from '../data/praticien.ts';

const NBSP = '\u00A0';
const ORDRE_JOURS: Jour[] = ['Lundi', 'Mardi', 'Mercredi', 'Jeudi', 'Vendredi', 'Samedi', 'Dimanche'];

export function formatPrix(euros: number): string {
  return `${euros}${NBSP}€`;
}

export function formatDuree(minutes: number): string {
  if (minutes < 60) return `${minutes}${NBSP}min`;
  const heures = Math.floor(minutes / 60);
  const reste = minutes % 60;
  return reste === 0 ? `${heures}${NBSP}h` : `${heures}${NBSP}h${NBSP}${String(reste).padStart(2, '0')}`;
}

export function formatHeure(hhmm: string): string {
  const [h, m] = hhmm.split(':');
  return m === '00' ? `${Number(h)}h` : `${Number(h)}h${m}`;
}

export function adresseComplete(lieu: Lieu): string {
  return `${lieu.rue}, ${lieu.codePostal} ${lieu.ville}`;
}

export function itineraireUrl(lieu: Lieu): string {
  return `https://www.google.com/maps/dir/?api=1&destination=${encodeURIComponent(adresseComplete(lieu))}`;
}

export function formatListe(elements: string[]): string {
  if (elements.length <= 1) return elements.join('');
  return `${elements.slice(0, -1).join(', ')} ou ${elements.at(-1)}`;
}

export function joursOuverts(lieu: Lieu): Jour[] {
  const jours = new Set(lieu.horaires.map((c) => c.jour));
  return ORDRE_JOURS.filter((j) => jours.has(j));
}
```

```ts
  nuit: '#14284B',

  surface: '#FFFFFF',

export const polices = {
  titre: "'Archivo Variable', 'Arial Narrow', sans-serif",
  texte: "'Source Sans 3 Variable', 'Segoe UI', sans-serif",
} as const;

function canal(valeur: number): number {
  const c = valeur / 255;
  return c <= 0.04045 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4;
}

function luminance(hex: string): number {
  const n = parseInt(hex.slice(1), 16);
  return 0.2126 * canal((n >> 16) & 255) + 0.7152 * canal((n >> 8) & 255) + 0.0722 * canal(n & 255);
}

export function ratioContraste(a: string, b: string): number {
  const [clair, fonce] = [luminance(a), luminance(b)].sort((x, y) => y - x);
  return (clair + 0.05) / (fonce + 0.05);
}
```

```ts
import { A_FOURNIR, praticien, type Jour, type Lieu } from '../data/praticien.ts';

const JOURS_SCHEMA: Record<Jour, string> = {
  Lundi: 'https://schema.org/Monday',
  Mardi: 'https://schema.org/Tuesday',
  Mercredi: 'https://schema.org/Wednesday',
  Jeudi: 'https://schema.org/Thursday',
  Vendredi: 'https://schema.org/Friday',
  Samedi: 'https://schema.org/Saturday',
  Dimanche: 'https://schema.org/Sunday',
};

const ID_PRATICIEN = `${praticien.siteUrl}/#praticien`;

export function schemaPraticien(): object {
  return {
    '@type': 'Person',
    '@id': ID_PRATICIEN,
    name: `${praticien.prenom} ${praticien.nom}`,
    jobTitle: praticien.titre,
    telephone: praticien.telephoneE164,
    email: `mailto:${praticien.email}`,
    url: `${praticien.siteUrl}/`,
  };
}

export function schemaLieu(lieu: Lieu): object {
  const prix = lieu.tarifs.map((t) => t.prixEuros);
  const min = Math.min(...prix);
  const max = Math.max(...prix);
  return {
    '@type': 'MedicalBusiness',
    '@id': `${praticien.siteUrl}/${lieu.slug}/#cabinet`,
    name: `${praticien.prenom} ${praticien.nom}, ${praticien.titre} à ${lieu.ville}`,
    url: `${praticien.siteUrl}/${lieu.slug}/`,
    telephone: praticien.telephoneE164,
    image: `${praticien.siteUrl}/images/logo-carre-512.png`,
    address: {
      '@type': 'PostalAddress',
      streetAddress: lieu.rue,
      postalCode: lieu.codePostal,
      addressLocality: lieu.ville,
      addressCountry: 'FR',
    },
    openingHoursSpecification: lieu.horaires.map((c) => ({
      '@type': 'OpeningHoursSpecification',
      dayOfWeek: JOURS_SCHEMA[c.jour],
      opens: c.ouverture,
      closes: c.fermeture,
    })),
    priceRange: min === max ? `${min} €` : `${min}-${max} €`,
    paymentAccepted: praticien.moyensPaiement.join(', '),
    employee: { '@id': ID_PRATICIEN },
    sameAs: [praticien.doctolibUrl, lieu.ficheGoogleUrl].filter((u) => u !== A_FOURNIR),
  };
}

export function grapheJsonLd(noeuds: object[]): { '@context': string; '@graph': object[] } {
  return { '@context': 'https://schema.org', '@graph': noeuds };
}
```

```ts
import assert from 'node:assert/strict';
import { existsSync } from 'node:fs';
import { join } from 'node:path';
import { DIST, PAGES, extraire, lirePage } from './outils.ts';

function cible(href: string): string {
  const chemin = href.split('#')[0];
  if (chemin === '/' || chemin === '') return 'index.html';
  return chemin.endsWith('/') ? `${chemin.slice(1)}index.html` : chemin.slice(1);
}

for (const page of PAGES) {
  test(`${page} : tous les liens internes existent`, () => {
    const liens = extraire(lirePage(page), /(?:href|src)="(\/(?!\/)[^"]*)"/g);
    const casses = liens.filter((l) => !existsSync(join(DIST, cible(l))));
    assert.deepEqual(casses, []);
  });
}
```

- [ ] **Step 4: Créer `site/scripts/verifier-lancement.mjs`**

```js
import { existsSync, readdirSync, readFileSync, statSync } from 'node:fs';
import { join } from 'node:path';

const MARQUEUR = '[À FOURNIR]';
const FICHIERS_FOURNIS = ['public/images/portrait.jpg', 'public/images/partenaires/gym-concordia.png'];

function pagesHtml(dossier) {
  return readdirSync(dossier).flatMap((nom) => {
    const chemin = join(dossier, nom);
    if (statSync(chemin).isDirectory()) return pagesHtml(chemin);
    return chemin.endsWith('.html') ? [chemin] : [];
  });
}

if (!existsSync('dist')) {
  console.error("Lancez d'abord : npm run build");
  process.exit(1);
}

const problemes = [];
for (const page of pagesHtml('dist')) {
  const nombre = readFileSync(page, 'utf8').split(MARQUEUR).length - 1;
  if (nombre > 0) problemes.push(`${page} : ${nombre} élément(s) à fournir`);
}
for (const fichier of FICHIERS_FOURNIS) {
  if (!existsSync(fichier)) problemes.push(`Fichier manquant : ${fichier}`);
}

if (problemes.length > 0) {
  console.error(`Mise en ligne bloquée :\n- ${problemes.join('\n- ')}`);
  process.exit(1);
}
console.log('Prêt pour la mise en ligne.');
```
